                'go': 'go'
            }
            file_language = language_map.get(file_ext, file_ext)
            
            # Apply each rule to the file
            for rule in self.rules:
                # Skip if rule doesn't apply to this language
                if 'languages' in rule and file_language not in rule['languages']:
                    continue
                
                # Skip if pattern failed to compile
                if rule.get('compiled_pattern') is None:
                    continue
                
                # Find all matches using regex
                matches = self._find_pattern_matches(content, lines, rule)
                
                # For each match, get context and confirm with Bob
                for match_info in matches:
                    violation = self._confirm_with_bob(
                        file_path=file_path,
                        match_info=match_info,
                        rule=rule,
                        lines=lines
                    )
                    
                    if violation:
                        violations.append(violation)
            
        except Exception as e:
            logger.error(f"Error analyzing file {file_path}: {e}")

    def _read_file_safely(self, file_path: Path) -> Optional[str]:
        """
        Safely read file content with encoding fallback.
        
        Args:
            file_path: Path to file to read
            
        Returns:
            File content as string, or None if reading fails
        """
        encodings = ['utf-8', 'latin-1', 'cp1252']
        
        for encoding in encodings:
            try:
                with open(file_path, 'r', encoding=encoding) as f:
                    return f.read()
            except UnicodeDecodeError:
                continue
            except Exception as e:
                logger.error(f"Error reading file {file_path}: {e}")
                return None
        
        logger.warning(f"Could not read file {file_path} with any supported encoding")
        return None
    
    def _find_pattern_matches(self, content: str, lines: List[str], 
                             rule: Dict[str, Any]) -> List[Dict[str, Any]]:
        """
        Find all matches of a rule's pattern in the content.
        
        Args:
            content: Full file content
            lines: Content split into lines
            rule: Rule dictionary with compiled pattern
            
        Returns:
            List of match information dictionaries
        """
        matches = []
        pattern = rule['compiled_pattern']
        
        for match in pattern.finditer(content):
            # Calculate line number from match position
            line_number = content[:match.start()].count('\n') + 1
            
            # Get the matched text
            matched_text = match.group(0)
            
            matches.append({
                'line_number': line_number,
                'matched_text': matched_text,
                'start_pos': match.start(),
                'end_pos': match.end()
            })
        
        return matches

<content># Analyzer Module Documentation

## Overview

The `analyzer.py` module is the core component of Bob-Guard that scans legacy codebases for compliance and security violations. It uses a two-stage detection approach:

1. **Fast Regex Detection**: Quickly scans files using regex patterns
2. **AI Confirmation**: Sends suspicious code to IBM Bob for deep analysis and confirmation

## Architecture

### Classes

#### `Violation`
Represents a detected code violation with all relevant details.

**Attributes:**
- `id`: Unique identifier for the violation
- `file_path`: Relative path to the file
- `line_number`: Line number where violation occurs
- `column`: Column number (optional)
- `rule_id`: ID of violated rule (e.g., 'SEC-001')
- `rule_name`: Human-readable rule name
- `category`: SECURITY, GDPR, HIPAA, or QUALITY
- `severity`: CRITICAL, HIGH, MEDIUM, or LOW
- `code_excerpt`/`code_snippet`: The problematic code
- `explanation`/`description`: Bob's detailed explanation
- `remediation_hint`/`recommendation`: Suggested fix
- `confidence`: Confidence score (0.0 to 1.0)
- `context_lines`: Surrounding code lines
- `cwe_id`: Common Weakness Enumeration ID (optional)
- `cvss_score`: CVSS score (optional)

#### `AnalysisResult`
Container for complete analysis results.

**Attributes:**
- `violations`: List of Violation objects
- `repository_path`: Path to analyzed repository
- `scan_timestamp`: When scan was performed
- `total_files_scanned`: Number of files analyzed
- `total_files`: Total files in repository
- `analyzed_files`: Files actually analyzed
- `rules_applied`: Number of rules checked
- `statistics`/`summary`: Statistical summary

#### `Analyzer`
Main analyzer class that orchestrates the scanning process.

**Key Features:**
- Recursive repository traversal
- File type filtering (`.py`, `.js`, `.java`, `.ts`, `.php`, `.rb`, `.go`)
- Directory exclusion (`node_modules`, `.git`, `__pycache__`, etc.)
- Regex pattern matching
- IBM Bob AI integration
- Error handling and logging

## Configuration

### Supported File Extensions

The analyzer scans files with these extensions:
- `.py` - Python
- `.js`, `.jsx` - JavaScript
- `.ts`, `.tsx` - TypeScript
- `.java` - Java
- `.php` - PHP
- `.rb` - Ruby
- `.go` - Go

To modify, update `Analyzer.SUPPORTED_EXTENSIONS`:

Analyzer.SUPPORTED_EXTENSIONS = {'.py', '.js', '.java'}

### Ignored Directories

By default, these directories are skipped:
- `node_modules`
- `.git`
- `__pycache__`
- `venv`, `env`, `.venv`
- `dist`, `build`
- `target`, `bin`, `obj`
- `.pytest_cache`, `.mypy_cache`
- `coverage`, `.tox`

To modify, update `Analyzer.IGNORED_DIRECTORIES`:

Analyzer.IGNORED_DIRECTORIES.add('my_custom_dir')

## Error Handling

The analyzer handles various error scenarios:

### File Reading Errors
```python
# Tries multiple encodings: utf-8, latin-1, cp1252
content = analyzer._read_file_safely(file_path)
if content is None:
    # File couldn't be read, skip it
    pass
```

### Bob API Errors
```python
try:
    violation = analyzer._confirm_with_bob(...)
except BobAPIError as e:
    # Falls back to regex-only detection with lower confidence
    logger.warning(f"Bob API error: {e}")
```

### Pattern Compilation Errors
```python
# Invalid regex patterns are logged and skipped
for rule in rules:
    try:
        rule['compiled_pattern'] = re.compile(rule['pattern'])
    except re.error as e:
        logger.warning(f"Invalid pattern in rule {rule['id']}: {e}")
```

## Performance Optimization

### Batch Processing
```python
# Process files in batches to manage memory
batch_size = 50
files = list(analyzer._get_scannable_files())

for i in range(0, len(files), batch_size):
    batch = files[i:i+batch_size]
    for file_path in batch:
        violations.extend(analyzer.analyze_file(file_path))
```

### Parallel Processing
```python
from concurrent.futures import ThreadPoolExecutor

def analyze_file_wrapper(file_path):
    return analyzer.analyze_file(file_path)

with ThreadPoolExecutor(max_workers=4) as executor:
    results = executor.map(analyze_file_wrapper, files)
    violations = [v for result in results for v in result]
```

## Best Practices

1. **Always use context managers with BobClient**
   ```python
   with BobClient() as bob_client:
       analyzer = Analyzer(repo_path, rules, bob_client)
       violations = analyzer.scan_repository()
   ```

2. **Log progress for large repositories**
   ```python
   import logging
   logging.basicConfig(level=logging.INFO)
   ```

3. **Handle violations incrementally**
   ```python
   for file_path in analyzer._get_scannable_files():
       violations = analyzer.analyze_file(file_path)
       # Process violations immediately
       process_violations(violations)
   ```

4. **Filter false positives**
   ```python
   # Only keep high-confidence violations
   confirmed = [v for v in violations if v.confidence >= 0.8]
   ```

5. **Save intermediate results**
   ```python
   # Save after each file or batch
   with open('progress.json', 'w') as f:
       json.dump([v.to_dict() for v in violations], f)
   ```

with open('config/rules.json', 'r') as f:

files = list(analyzer._get_scannable_files())[:100]
